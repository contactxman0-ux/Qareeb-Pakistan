const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; const OSM_TAGS: Record = { restaurants: '["amenity"~"restaurant|fast_food|food_court|eating_house"]', dhabas: '["amenity"~"cafe|tea_house|dhaba"]', clinics: '["amenity"~"clinic|hospital|doctors|health_post|pharmacy"]', pharmacy: '["amenity"="pharmacy"]', grocery: '["shop"~"supermarket|grocery|general|convenience|department_store"]', salons: '["shop"~"hairdresser|beauty|cosmetics|barber"]', atm: '["amenity"~"atm|bank"]', mosque: '["amenity"="place_of_worship"]["religion"="muslim"]', petrol: '["amenity"="fuel"]', }; Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { const { lat, lng, category, radius } = await req.json(); const r = radius || 15000; const tag = OSM_TAGS[category] || '["amenity"="restaurant"]'; const query = `[out:json][timeout:30];(node${tag}(around:${r},${lat},${lng});way${tag}(around:${r},${lat},${lng}););out body center 25;`; const res = await fetch("https://overpass-api.de/api/interpreter", { method: "POST", body: "data=" + encodeURIComponent(query), headers: { "Content-Type": "application/x-www-form-urlencoded" }, }); const data = await res.json(); const elements = data.elements || []; // Convert OSM format to Google Places-like format const seen = new Set(); const results = []; for (const el of elements) { const name = el.tags?.name || el.tags?.["name:en"] || el.tags?.["name:ur"]; if (!name || seen.has(name)) continue; seen.add(name); const elat = el.lat ?? el.center?.lat ?? lat; const elng = el.lon ?? el.center?.lon ?? lng; const phone = el.tags?.["contact:phone"] || el.tags?.phone || ""; const addr = el.tags?.["addr:street"] || el.tags?.["addr:city"] || el.tags?.["addr:full"] || ""; results.push({ place_id: `osm_${el.id}`, name: name, vicinity: addr || name, geometry: { location: { lat: elat, lng: elng } }, rating: 0, user_ratings_total: 0, opening_hours: { open_now: true }, formatted_phone_number: phone, }); if (results.length >= 20) break; } return new Response(JSON.stringify({ results, status: "OK" }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } catch (e) { return new Response( JSON.stringify({ error: e.message, results: [], status: "ERROR" }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 200 } ); } });
